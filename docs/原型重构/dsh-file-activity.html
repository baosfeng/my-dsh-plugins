TS view.ts 
import { type Context } from 'cordis';
import { renderRows } from './rows';
import { renderTree } from './tree';
import { openPreview } from './preview-float';

export function createFileActivityView(ctx) {
  const root = document.createElement('div');
  root.className = 'fa-panel';

  const recent = renderRows(ctx, { limit: 5, dedupe: true });
  const tree = renderTree(ctx, { compressChain: true });

  root.append(recent.el, tree.el);
  root.addEventListener('click', (event) => {
    const row = event.target.closest('[data-file]');
    if (row) openPreview(ctx, { path: row.dataset.path });
  });

  return { el: root, dispose: () => root.remove() };
}

dsh-file-activity

右侧栏文件活动面板：记录 agent 读写过的文件，按操作类型与路径聚合展示。

  • 最近访问最多 5 条，同一文件只保留最近一次。
  • 文件统计按目录聚合成树，单链目录压缩显示。
  • 点击任意文件打开浮窗预览，支持代码 / Markdown / 图片 / PDF / 沙箱 HTML。

安装

npm install dsh-file-activity

记录只保存在内存中，重启后从空态开始。
