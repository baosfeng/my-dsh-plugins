实现方案

把重试、超时、取消拆成三个正交的部分，外层只负责编排。核心是 fetchWithRetry：每次尝试独立计时，失败后按退避策略等待，并在等待期间响应取消。

  • 可重试判定：网络错误、408、429、5xx；其余错误直接抛出。
  • 退避方式：指数退避 + 全抖动（Math.random()），上限 maxDelay。
  • 超时：单次尝试超时，不做整体超时，避免重试被总时长截断。

export async function fetchWithRetry(url, opts = {}) {
  const { retries = 3, baseDelay = 300, maxDelay = 8000,
    timeout = 10000, signal, ...init } = opts;

  for (let attempt = 0; ; attempt += 1) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(new TimeoutError(url, timeout)), timeout);
    const merged = signal ? AbortSignal.any([signal, ctrl.signal]) : ctrl.signal;

    try {
      const res = await fetch(url, { ...init, signal: merged });
      if (res.ok || !isRetryableStatus(res.status)) return res;
    } catch (error) {
      if (attempt >= retries || !isRetryableError(error)) throw error;
    } finally {
      clearTimeout(timer);
    }

    const delay = Math.min(maxDelay, baseDelay * 2 ** attempt);
    await sleep(delay * Math.random(), signal);
  }
}

审查结论：重试主流程可以通过，但有两处需要修。

  • sleep 未响应 signal，取消时最多多等 8 秒。
  • isRetryableStatus 漏了 408。
TS 建议改法 
function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    const id = setTimeout(resolve, ms);
    signal?.addEventListener('abort', () => {
      clearTimeout(id);
      reject(signal.reason);
    }, { once: true });
  });
}
